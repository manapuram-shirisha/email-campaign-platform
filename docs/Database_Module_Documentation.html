Objective

Configure the project database layer so the platform has a PostgreSQL database, a Prisma data model, generated Prisma Client, applied schema migration, and demo seed data for the next modules.

Why This Module Matters

Persistent Data

Contacts, templates, campaigns, events, suppressions, users, and organisations all require a relational database.

Demo Readiness

Seed data gives the app a super admin, contacts, templates, segment, and campaign draft for testing.

Issues Found and Fixed

Prisma auto-install failed: Prisma attempted to run pnpm add prisma@6.19.3 automatically. We fixed this by installing Prisma and Prisma Client manually.
DATABASE_URL not found: Prisma commands run from apps/api could not see the root .env. We fixed this with dotenv-cli in Prisma scripts.
Prisma Studio env conflict: Both root .env and prisma/.env had DATABASE_URL. We consolidated to the root .env.
Authentication failed: PostgreSQL credentials needed to match the actual local server and port. The final working URL uses localhost:5433.
Empty database in Studio: Tables existed but counts were zero before seed ran successfully. The seed script populated the required demo data.

model User {
  id           String   @id @default(cuid())
  email        String   @unique
  name         String?
  passwordHash String
  role         UserRole @default(CAMPAIGN_MANAGER)
  orgId        String
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  organisation Organisation @relation(fields: [orgId], references: [id], onDelete: Cascade)
}

model ContactList {
  id          String   @id @default(cuid())
  orgId       String
  name        String
  description String?
  tags        String[] @default([])
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  organisation Organisation        @relation(fields: [orgId], references: [id], onDelete: Cascade)
  members      ContactListMember[]
  importJobs   ImportJob[]

  @@unique([orgId, name])
}

model Contact {
  id           String        @id @default(cuid())
  orgId        String
  email        String
  firstName    String?
  lastName     String?
  phone        String?
  status       ContactStatus @default(ACTIVE)
  customFields Json          @default("{}")
  source       ContactSource @default(MANUAL)
  createdAt    DateTime      @default(now())
  updatedAt    DateTime      @updatedAt

  organisation Organisation        @relation(fields: [orgId], references: [id], onDelete: Cascade)
  lists        ContactListMember[]
  sends        CampaignSend[]
  events       EmailEvent[]

  @@unique([orgId, email])
}

model ContactListMember {
  contactId    String
  listId       String
  subscribedAt DateTime @default(now())

  contact Contact     @relation(fields: [contactId], references: [id], onDelete: Cascade)

  @@id([contactId, listId])
}

  orgId        String
  name         String

  fromName     String

  replyToEmail String?
  status       CampaignStatus @default(DRAFT)

  createdAt    DateTime       @default(now())
  updatedAt    DateTime       @updatedAt

  organisation Organisation   @relation(fields: [orgId], references: [id], onDelete: Cascade)

model CampaignSend {
  id           String     @id @default(cuid())
  campaignId   String
  contactId    String
  sesMessageId String?
  status       SendStatus @default(QUEUED)
  sentAt       DateTime?
  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt

  campaign Campaign @relation(fields: [campaignId], references: [id], onDelete: Cascade)
  contact  Contact  @relation(fields: [contactId], references: [id], onDelete: Cascade)

  @@unique([campaignId, contactId])
}

model EmailEvent {
  id         String    @id @default(cuid())
  contactId String?
  campaignId String?
  eventType EventType
  occurredAt DateTime @default(now())
  ip         String?
  userAgent  String?
  metadata   Json      @default("{}")

  contact  Contact?  @relation(fields: [contactId], references: [id], onDelete: SetNull)
  campaign Campaign? @relation(fields: [campaignId], references: [id], onDelete: SetNull)

  @@index([campaignId, eventType])
  @@index([contactId, eventType])
}

model SuppressionList {
  id           String            @id @default(cuid())
  orgId        String
  email        String
  reason       SuppressionReason
  suppressedAt DateTime          @default(now())

  organisation Organisation @relation(fields: [orgId], references: [id], onDelete: Cascade)

  @@unique([orgId, email])
}

model ImportJob {
  id        String       @id @default(cuid())
  listId    String
  status    ImportStatus @default(PENDING)
  totalRows Int          @default(0)
  added     Int          @default(0)
  updated   Int          @default(0)
  skipped   Int          @default(0)
  errored   Int          @default(0)
  startedAt DateTime?
  endedAt   DateTime?
  createdAt DateTime     @default(now())

  list ContactList @relation(fields: [listId], references: [id], onDelete: Cascade)
}

async function main() {
  const organisation = await prisma.organisation.upsert({
    where: {
      id: "seed-org"
    },
    update: {},
    create: {
      id: "seed-org",
      name: "Demo Organisation",
      fromEmail: "verified@example.com",
      sesConfigSet: "email-platform-events",
      awsRegion: "ap-south-1"
    }
  });

  const passwordHash = await bcrypt.hash("Admin@123", 10);

  await prisma.user.upsert({
    where: {
      email: "admin@example.com"
    },
    update: {
      passwordHash,
      role: UserRole.SUPER_ADMIN,
      orgId: organisation.id
    },
    create: {
      email: "admin@example.com",
      name: "Super Admin",
      passwordHash,
      role: UserRole.SUPER_ADMIN,
      orgId: organisation.id
    }
  });

  const list = await prisma.contactList.upsert({
    where: {
      orgId_name: {
        orgId: organisation.id,
        name: "Newsletter Subscribers"
      }
    },
    update: {},
    create: {
      orgId: organisation.id,
      name: "Newsletter Subscribers",
      description: "Seed list for demo contacts",
      tags: ["demo", "newsletter"]
    }
  });

  const contacts = [
    {
      email: "aisha@example.com",
      firstName: "Aisha",
      lastName: "Rao",
      customFields: {
        company: "Acme",
        city: "Chennai"
      }
    },
    {
      email: "rahul@example.com",
      firstName: "Rahul",
      lastName: "Mehta",
      customFields: {
        company: "Northstar",
        city: "Bengaluru"
      }
    },
    {
      email: "priya@example.com",
      firstName: "Priya",
      lastName: "Sharma",
      customFields: {
        company: "Bluepeak",
        city: "Mumbai"
      }
    }
  ];

  for (const item of contacts) {
    const contact = await prisma.contact.upsert({
      where: {
        orgId_email: {
          orgId: organisation.id,
          email: item.email
        }
      },
      update: {
        firstName: item.firstName,
        lastName: item.lastName,
        customFields: item.customFields
      },
      create: {
        orgId: organisation.id,
        email: item.email,
        firstName: item.firstName,
        lastName: item.lastName,
        source: "MANUAL",
        customFields: item.customFields
      }
    });

    await prisma.contactListMember.upsert({
      where: {
        contactId_listId: {
          contactId: contact.id,
          listId: list.id
        }
      },
      update: {},
      create: {
        contactId: contact.id,
        listId: list.id
      }
    });
  }

  const createdTemplates = [];

  for (const template of starterTemplates) {
    const created = await prisma.template.upsert({
      where: {
        id: `seed-template-${template.name.toLowerCase().replaceAll(" ", "-")}`
      },
      update: {
        name: template.name,
        category: template.category,
        html: template.html,
        blocks: []
      },
      create: {
        id: `seed-template-${template.name.toLowerCase().replaceAll(" ", "-")}`,
        orgId: organisation.id,
        name: template.name,
        category: template.category,
        html: template.html,
        blocks: []
      }

    createdTemplates.push(created);
